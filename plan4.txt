Wait, let's write a function `getRealisticScale(type)`.
```javascript
window.getRealisticScale = function(type) {
    if (!type) return 1;
    // 40px = 4 feet, 10px = 1 foot.
    // Base furniture size is 100x100px = 10x10 feet.
    // To make it X feet, scale should be (X * 10) / 100 = X / 10.
    if (type.includes('chair') || type.includes('nightstand') || type.includes('bin') || type.includes('succulent') || type.includes('holorose')) return 0.25; // 2.5 ft
    if (type.includes('plant') || type.includes('fern') || type.includes('cactus') || type.includes('bonsai') || type.includes('orchid') || type.includes('ficus') || type.includes('monstera') || type.includes('bamboo') || type.includes('mushroom') || type.includes('sunflower') || type.includes('lunarlily') || type.includes('lavalamp') || type.includes('plasmaglobe') || type.includes('lamp')) return 0.3; // 3 ft
    if (type.includes('table') || type.includes('desk') || type.includes('cabinet') || type.includes('gamingpc') || type.includes('tv') || type.includes('arcade') || type.includes('display') || type.includes('fridge') || type.includes('server') || type.includes('shelf') || type.includes('wardrobe') || type.includes('smartmirror') || type.includes('cyberposter') || type.includes('neonsign')) return 0.4; // 4 ft
    if (type.includes('bench')) return 0.5; // 5 ft
    if (type.includes('bed')) return 0.7; // 7 ft
    if (type.includes('tree') || type.includes('pine')) return 0.8; // 8 ft
    if (type.includes('carpet')) return 0.9; // 9 ft
    return 0.5;
};
```
