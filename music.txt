window.WildPulseMusic = {
    initialized: false,
    muted: false,
    state: 'MENU',
    bpm: 100,
    synths: {},
    loops: [],
    synth: null, // Legacy animations fallback

    async start() {
        if (typeof Tone !== 'undefined' && Tone.context && Tone.context.state === 'suspended') {
            try { await Tone.context.resume(); } catch(e) {}
        }
        if (this.initialized) return;
        try {
            Tone.setContext(new Tone.Context({ latencyHint: "playback" }));
            await Tone.start();
            Tone.getContext().lookAhead = 0.5;
            Tone.getContext().updateInterval = 0.05;
            this.initialized = true;

            // Master volume bus
            this.musicBus = new Tone.Volume(-4).toDestination();

            // Legacy PolySynth to prevent errors
            this.synth = new Tone.PolySynth(Tone.Synth).connect(this.musicBus);

            // shared Reverb and Delay
            this.reverb = new Tone.Reverb(3.5).connect(this.musicBus);
            this.reverb.wet.value = 0.45;

            // Atmospheric setup

            // Wind
            this.windFilter = new Tone.Filter({ type: "lowpass", frequency: 400, rolloff: -24 }).connect(this.reverb);
            this.windLFO = new Tone.LFO({ type: "sine", min: 200, max: 800, frequency: 0.1 }).start();
            this.windLFO.connect(this.windFilter.frequency);
            this.synths.wind = new Tone.NoiseSynth({ noise: { type: 'brown' }, envelope: { attack: 4, decay: 0.2, sustain: 1, release: 4 } }).connect(this.windFilter);
            this.synths.wind.volume.value = -60;

            // Rain
            this.rainFilter = new Tone.Filter({ type: "bandpass", frequency: 1000, Q: 0.5 }).connect(this.reverb);
            this.synths.rain = new Tone.NoiseSynth({ noise: { type: 'pink' }, envelope: { attack: 2, decay: 0.1, sustain: 1, release: 2 } }).connect(this.rainFilter);
            this.synths.rain.volume.value = -60;

            // Thunder
            this.synths.thunderStrike = new Tone.MembraneSynth({ pitchDecay: 0.05, octaves: 4, oscillator: { type: "square" }, envelope: { attack: 0.001, decay: 0.4, sustain: 0.01, release: 1.4 } }).connect(this.reverb);
            this.synths.thunderStrike.volume.value = -5;
            let thunderDist = new Tone.Distortion(0.8).connect(new Tone.Filter(300, "lowpass").connect(this.reverb));
            this.synths.thunder = new Tone.NoiseSynth({ noise: { type: 'brown' }, envelope: { attack: 0.1, decay: 1, sustain: 0.5, release: 4 } }).connect(thunderDist);
            this.synths.thunder.volume.value = -5;

            // Birds (Daytime)
            this.synths.birds = new Tone.Synth({
                oscillator: { type: "sine" },
                envelope: { attack: 0.05, decay: 0.1, sustain: 0.1, release: 0.5 }
            }).connect(this.reverb);
            this.synths.birds.volume.value = -10;

            // Crickets (Nighttime)
            this.synths.crickets = new Tone.NoiseSynth({
                noise: { type: 'white' },
                envelope: { attack: 0.01, decay: 0.05, sustain: 0, release: 0.05 }
            }).connect(new Tone.Filter(4000, "highpass").connect(this.reverb));
            this.synths.crickets.volume.value = -25;

            // Player Steps
            this.stepFilter = new Tone.Filter({ type: "lowpass", frequency: 1200 }).connect(this.reverb);
            this.synths.steps = new Tone.NoiseSynth({ noise: { type: 'pink' }, envelope: { attack: 0.01, decay: 0.1, sustain: 0, release: 0.1 } }).connect(this.stepFilter);
            this.synths.steps.volume.value = -12;

            // Leaves Rustling
            this.leafFilter = new Tone.Filter({ type: "bandpass", frequency: 2000, Q: 1.5 }).connect(this.reverb);
            this.synths.leaves = new Tone.NoiseSynth({ noise: { type: 'white' }, envelope: { attack: 0.1, decay: 0.3, sustain: 0, release: 0.5 } }).connect(this.leafFilter);
            this.synths.leaves.volume.value = -35;

            // Owls (Nighttime)
            this.synths.owl = new Tone.Synth({
                oscillator: { type: "sine" },
                envelope: { attack: 0.1, decay: 0.2, sustain: 0.8, release: 1 }
            }).connect(this.reverb);
            this.synths.owl.volume.value = -15;

            this.setupLoops();
            Tone.Transport.bpm.value = this.bpm;
            Tone.Transport.start();
            console.log("[Music] Advanced procedural ambient sound engine started.");
        } catch (e) {
            console.warn("[Music] Failed to start Tone.js:", e);
        }
    },

    setupLoops() {
        this.loops.push(new Tone.Loop(time => {
            const step = Math.round((Tone.Transport.getTicksAtTime(time) / Tone.Transport.PPQ) * 4) % 16;
            const now = Date.now();
            const dayNight = (typeof dayNightTime !== 'undefined') ? dayNightTime : 12.0;
            const isNight = (dayNight < 6 || dayNight >= 18);
            const isDay = !isNight;
            const inside = window.isPlayerInsideHome ? window.isPlayerInsideHome(1) : false;

            if (inside) return;

            // Day: Birds
            if (isDay && (window.weatherPattern === 'none' || !window.weatherPattern) && Math.random() < 0.1) {
                const note = ["C6", "E6", "G6", "C7"][Math.floor(Math.random() * 4)];
                this.synths.birds.triggerAttackRelease(note, "16n", time, Math.random() * 0.5 + 0.5);
                if (Math.random() < 0.5) {
                     this.synths.birds.triggerAttackRelease("E6", "16n", time + 0.1, Math.random() * 0.5 + 0.5);
                }
            }

            // Night: Crickets & Owls
            if (isNight) {
                if (step % 2 === 0 && Math.random() < 0.8) {
                    this.synths.crickets.triggerAttackRelease("32n", time, 0.5);
                }

                if (step === 0 && Math.random() < 0.02) {
                    this.synths.owl.triggerAttackRelease("C4", "4n", time, 0.8);
                    this.synths.owl.triggerAttackRelease("G3", "2n", time + 0.5, 0.6);
                }
            }
        }, "16n").start(0));
    },

    update() {
        if (!this.initialized || this.muted) return;

        if (typeof Tone !== 'undefined') {
            if (Tone.context && Tone.context.state === 'suspended') {
                try { Tone.context.resume(); } catch(e) {}
            }
            if (Tone.Transport && Tone.Transport.state !== 'started') {
                try { Tone.Transport.start(); } catch(e) {}
            }
        }

        const now = Date.now();
        const isInsideHome = window.isPlayerInsideHome ? window.isPlayerInsideHome(1) : false;

        // Fast-update precise sound effects
        if (!isInsideHome) {
            let p1Moving = (typeof window.playerVx !== 'undefined' && (window.playerVx !== 0 || window.playerVy !== 0));
            if (p1Moving) {
                if (!this.lastStepTime || now - this.lastStepTime > 250) {
                    if (this.synths.steps) {
                        this.synths.steps.triggerAttackRelease("16n", Tone.now(), 0.5);
                    }
                    this.lastStepTime = now;
                }
            } else {
                this.lastStepTime = 0;
            }
        }

        if (this.lastMusicUpdate && now - this.lastMusicUpdate < 200) return;
        this.lastMusicUpdate = now;

        // Sound effects
        if (!isInsideHome) {
            if (this.synths.leaves && window.windSpeed > 100 && Math.random() < (window.windSpeed / 2000)) {
                this.synths.leaves.triggerAttackRelease("8n", Tone.now(), Math.random() * 0.3 + 0.2);
            }
        }

        const weather = window.weatherPattern || 'none';
        const hasWeather = (weather !== 'none');
        const intensity = window.weatherIntensity || 0;
        const inside = window.isPlayerInsideHome ? window.isPlayerInsideHome(1) : false;

        let targetWindVol = -60;
        let targetRainVol = -60;

        if (this.reverb) {
            const targetWet = hasWeather ? 0.35 + (0.45 * intensity) : 0.35;
            if (Math.abs(this.reverb.wet.value - targetWet) > 0.05) {
                this.reverb.wet.rampTo(targetWet, 4);
            }
        }

        if (!inside) {
            if (weather !== 'none' && intensity > 0) {
                const baseWind = -45;
                targetWindVol = baseWind + (20 * intensity);

                if (this.windLFO) {
                    this.windLFO.frequency.value = 0.05 + (window.windSpeed / 500);
                }
            }

            if ((weather.includes('rain') || weather.includes('storm') || weather.includes('snow')) && intensity > 0) {
                const baseRain = -40;
                const maxRain = (weather === 'thunderstorm') ? -15 : -25;
                targetRainVol = baseRain + ((maxRain - baseRain) * intensity);
            }
        }

        if (this.synths.wind) {
            if (Math.abs(this.synths.wind.volume.value - targetWindVol) > 1) {
                this.synths.wind.volume.rampTo(targetWindVol, 5);
            }
            if (targetWindVol > -58 && this.synths.wind.state !== 'started') {
                this.synths.wind.triggerAttack();
            } else if (targetWindVol <= -58 && this.synths.wind.state === 'started') {
                this.synths.wind.triggerRelease();
            }
            if (this.synths.wind.state === 'started') {
                if (window.windSpeed > 300) {
                     this.synths.wind.noise.type = 'pink';
                } else {
                     this.synths.wind.noise.type = 'brown';
                }
            }
        }

        if (this.synths.rain) {
            const variation = (Math.random() - 0.5) * 2;
            const finalRainVol = targetRainVol + variation;

            if (Math.abs(this.synths.rain.volume.value - finalRainVol) > 1) {
                this.synths.rain.volume.rampTo(finalRainVol, 5);
            }
            if (targetRainVol > -58) {
                if (this.synths.rain.state !== 'started') this.synths.rain.triggerAttack();
                if (weather.includes('thunderstorm')) {
                    this.synths.rain.noise.type = 'white';
                } else if (weather.includes('storm')) {
                    this.synths.rain.noise.type = 'pink';
                } else if (weather.includes('snow')) {
                    this.synths.rain.noise.type = 'brown';
                } else {
                    this.synths.rain.noise.type = 'pink';
                }
            } else if (this.synths.rain.state === 'started') {
                this.synths.rain.triggerRelease();
            }
        }
    },

    triggerThunder() {
        if (!this.initialized || this.muted || !this.synths.thunder) return;
        const inside = window.isPlayerInsideHome ? window.isPlayerInsideHome(1) : false;
        const volOffset = inside ? -10 : 0;
        if (this.synths.thunderStrike) {
            this.synths.thunderStrike.volume.value = -5 + volOffset;
            this.synths.thunderStrike.triggerAttackRelease("C1", "8n");
        }
        if (this.synths.thunder) {
            this.synths.thunder.volume.value = -5 + volOffset;
            this.synths.thunder.triggerAttackRelease("1n");
        }
    },

    toggle() {
        this.muted = !this.muted;
        if (typeof Tone !== 'undefined') {
            Tone.Destination.mute = this.muted;
            if (!this.muted && Tone.context && Tone.context.state === 'suspended') {
                try { Tone.context.resume(); } catch(e) {}
            }
        }
        return this.muted;
    }
};
